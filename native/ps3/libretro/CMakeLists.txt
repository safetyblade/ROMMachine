cmake_minimum_required(VERSION 3.22)
project(cartdeck_ps3_libretro LANGUAGES CXX)
if(NOT DEFINED RPCS3_LIBRETRO_HEADERS)
    message(FATAL_ERROR "RPCS3_LIBRETRO_HEADERS must point to pinned libretro-common/include")
endif()
add_library(cartdeckps3vulkanhost SHARED CartdeckVulkanPresenter.cpp WiiuLibretroRunner.cpp WiiuAudio.cpp)
target_include_directories(cartdeckps3vulkanhost PRIVATE "${RPCS3_LIBRETRO_HEADERS}")
target_compile_features(cartdeckps3vulkanhost PRIVATE cxx_std_17)
target_compile_options(cartdeckps3vulkanhost PRIVATE -Wall -Wextra)
target_link_options(cartdeckps3vulkanhost PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(cartdeckps3vulkanhost PRIVATE android vulkan log dl aaudio)
